cmake_minimum_required(VERSION 3.25)
project(FOMHD VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(FOMHD_ENABLE_CUDA "Build the explicit single-GPU backend" OFF)
option(FOMHD_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(fomhd_core src/core/backend.cpp src/core/state.cpp src/boundary/wall.cpp
    src/diagnostics/diagnostics.cpp src/diagnostics/source.cpp src/time/stage.cpp src/time/midpoint.cpp
    src/fluid/fluid.cpp src/fields/darwin.cpp)
target_link_libraries(fomhd_core PUBLIC fomhd_numerics)
target_compile_options(fomhd_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
if(FOMHD_ENABLE_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit 13.0 REQUIRED)
    target_sources(fomhd_core PRIVATE src/cuda/backend.cu)
    target_compile_definitions(fomhd_core PRIVATE FOMHD_HAS_CUDA=1)
    target_link_libraries(fomhd_core PRIVATE CUDA::cudart CUDA::cublas CUDA::cufft CUDA::cusolver)
    set_target_properties(fomhd_core PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES CUDA_ARCHITECTURES "80;120")
    target_compile_options(fomhd_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
if(FOMHD_SANITIZERS)
    foreach(t fomhd_numerics fomhd_core)
        target_compile_options(${t} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${t} PUBLIC -fsanitize=address,undefined)
    endforeach()
endif()
